03 Technical Implementation

Time-Series Feature Engineering

// PySpark - Predictive Maintenance Features 
from pyspark.sql.functions import *
from pyspark.sql.window import Window

def engineer_features(sensor_df, window_sizes=[1, 7, 30]):
    """Generate predictive maintenance features from sensor data"""

    features = sensor_df

    for window in window_sizes:
        w = Window.partitionBy("asset_id").orderBy("timestamp").rowsBetween(-window*24*6, 0)

        features = features.withColumn(
            f"vibration_mean_{window}d", avg("vibration").over(w)
        ).withColumn(
            f"vibration_std_{window}d", stddev("vibration").over(w)
        ).withColumn(
            f"vibration_max_{window}d", max("vibration").over(w)
        ).withColumn(
            f"temp_mean_{window}d", avg("temperature").over(w)
        ).withColumn(
            f"temp_trend_{window}d",
            (last("temperature").over(w) - first("temperature").over(w)) / window
        )

    # Rate of change features
    features = features.withColumn(
        "vibration_roc",
        (col("vibration") - lag("vibration", 6).over(
            Window.partitionBy("asset_id").orderBy("timestamp")
        )) / 6
    )

    # Anomaly detection features
    features = features.withColumn(
        "vibration_zscore",
        (col("vibration") - col("vibration_mean_30d")) / col("vibration_std_30d")
    )

    return features

LSTM Failure Prediction Model

// Python - SageMaker LSTM Training 
import tensorflow as tf
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import LSTM, Dense, Dropout

def build_failure_prediction_model(sequence_length, n_features):
    """Build LSTM model for remaining useful life prediction"""

    model = Sequential([
        LSTM(128, return_sequences=True,
             input_shape=(sequence_length, n_features)),
        Dropout(0.2),
        LSTM(64, return_sequences=True),
        Dropout(0.2),
        LSTM(32, return_sequences=False),
        Dropout(0.2),
        Dense(32, activation='relu'),
        Dense(1, activation='sigmoid')  # Failure probability
    ])

    model.compile(
        optimizer=tf.keras.optimizers.Adam(learning_rate=0.001),
        loss='binary_crossentropy',
        metrics=['accuracy', tf.keras.metrics.AUC(name='auc')]
    )

    return model
