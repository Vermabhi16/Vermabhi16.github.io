03 Technical Implementation

Identity Resolution

// PySpark - Probabilistic Identity Matching 
from graphframes import GraphFrame

def build_identity_graph(interactions_df):
    """Build identity graph and resolve to canonical customer IDs"""

    # Create edges from deterministic matches
    deterministic_edges = interactions_df.select(
        col("source_id").alias("src"),
        col("match_key").alias("dst"),
        lit("deterministic").alias("match_type"),
        lit(1.0).alias("confidence")
    ).filter(col("match_key").isNotNull())

    # Create edges from probabilistic matches
    probabilistic_edges = (interactions_df
        .alias("a")
        .join(interactions_df.alias("b"),
            (col("a.device_fingerprint") == col("b.device_fingerprint")) &
            (col("a.source_id") != col("b.source_id")))
        .select(
            col("a.source_id").alias("src"),
            col("b.source_id").alias("dst"),
            lit("probabilistic").alias("match_type"),
            calculate_match_score(col("a"), col("b")).alias("confidence")
        )
        .filter(col("confidence") > 0.85))

    # Combine edges
    all_edges = deterministic_edges.union(probabilistic_edges)

    # Build graph and find connected components
    vertices = interactions_df.select(col("source_id").alias("id")).distinct()
    g = GraphFrame(vertices, all_edges)

    # Connected components = customer clusters
    clusters = g.connectedComponents()

    return clusters.withColumnRenamed("component", "canonical_customer_id")

Feature Engineering

// PySpark - Customer Feature Generation 
def generate_customer_features(customer_id):
    """Generate 500+ features for each customer"""

    # RFM Features
    rfm_features = transactions.groupBy("customer_id").agg(
        datediff(current_date(), max("transaction_date")).alias("recency_days"),
        count("*").alias("frequency"),
        sum("amount").alias("monetary_value"),
        avg("amount").alias("avg_order_value"),
        stddev("amount").alias("order_value_stddev")
    )

    # Behavioral Features
    behavior_features = events.groupBy("customer_id").agg(
        count(when(col("event_type") == "view", 1)).alias("total_page_views"),
        count(when(col("event_type") == "cart_add", 1)).alias("total_cart_adds"),
        countDistinct("session_id").alias("total_sessions"),
        avg("session_duration").alias("avg_session_duration")
    )

    # Channel Preference Features
    channel_features = transactions.groupBy("customer_id").agg(
        sum(when(col("channel") == "online", col("amount"))).alias("online_spend"),
        sum(when(col("channel") == "store", col("amount"))).alias("store_spend"),
        count(when(col("channel") == "mobile", 1)).alias("mobile_transactions")
    )

    return rfm_features.join(behavior_features, "customer_id") \
                       .join(channel_features, "customer_id")

06 Key Learnings

  • Identity resolution is foundational - invest heavily in graph-based matching
  • Feature store is essential for consistent ML features across training and serving
  • Start with deterministic matching before probabilistic approaches
  • Privacy-by-design must be built in from the start, not bolted on
  • Real-time activation requires careful latency planning in feature serving
