from fuzzywuzzy import fuzz
from sklearn.ensemble import RandomForestClassifier
import recordlinkage

def match_entities(source_df, master_df, entity_type):
    # Create comparison pairs using blocking
    indexer = recordlinkage.Index()
    indexer.block(left_on='normalized_name_block', right_on='normalized_name_block')
    candidate_pairs = indexer.index(source_df, master_df)

    # Feature extraction
    compare = recordlinkage.Compare()
    compare.string('name', 'name', method='jarowinkler', label='name_score')
    compare.string('address', 'address', method='levenshtein', label='addr_score')
    compare.exact('postal_code', 'postal_code', label='postal_match')
    compare.string('tax_id', 'tax_id', method='exact', label='taxid_match')
    features = compare.compute(candidate_pairs, source_df, master_df)

    # ML-based classification
    model = load_trained_model(entity_type)
    match_probability = model.predict_proba(features)[:, 1]

    # Apply thresholds
    matches = features[match_probability > 0.85].copy()
    matches['probability'] = match_probability[match_probability > 0.85]
    matches['review_required'] = (matches['probability'] < 0.95)

    return matches

Survivorship Rules

// T-SQL - Golden Record Creation 
CREATE PROCEDURE mdm.CreateGoldenRecord
    @EntityType VARCHAR(50),
    @ClusterId BIGINT
AS
BEGIN
    -- Get all source records in cluster
    WITH SourceRecords AS (
        SELECT *, ROW_NUMBER() OVER (
            PARTITION BY cluster_id
            ORDER BY source_trust_score DESC, last_updated DESC
        ) as trust_rank
        FROM mdm.staging_records
        WHERE cluster_id = @ClusterId
    ),
    -- Apply survivorship rules per attribute
    GoldenAttributes AS (
        SELECT
            @ClusterId as golden_id,
            -- Most trusted source for legal name
            MAX(CASE WHEN source_system = 'SAP' THEN legal_name END) as legal_name,
            -- Most recent for address
            (SELECT TOP 1 address FROM SourceRecords ORDER BY last_updated DESC) as address,
            -- Most complete for phone
            MAX(CASE WHEN phone IS NOT NULL THEN phone END) as phone,
            -- Aggregated for classifications
            STRING_AGG(DISTINCT classification, '|') as classifications
        FROM SourceRecords
        GROUP BY cluster_id
    )
    MERGE mdm.golden_records AS target
    USING GoldenAttributes AS source
    ON target.golden_id = source.golden_id
    WHEN MATCHED THEN UPDATE SET
        legal_name = source.legal_name,
        address = source.address,
        updated_date = GETUTCDATE()
    WHEN NOT MATCHED THEN INSERT (golden_id, legal_name, address, phone)
    VALUES (source.golden_id, source.legal_name, source.address, source.phone);
END;
